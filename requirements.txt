#
#   Andes:    module load gcc
#             python3 -m venv ~/venvs/vod2uv && source ~/venvs/vod2uv/bin/activate
#             MPICC="mpicc -shared" pip install --no-cache-dir -r requirements.txt
#
#   Frontier: module load PrgEnv-gnu cray-python ; module unload rocm
#             python3 -m venv ~/venvs/vod2uv && source ~/venvs/vod2uv/bin/activate
#             MPICC="cc -shared" pip install --no-cache-dir -r requirements.txt
#
# Verify:  python -c "import mpi4py; print(mpi4py.get_config())"   (should show the Cray/system mpicc)
